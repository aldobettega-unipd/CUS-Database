UNIVERSITA (nome, citta)


PERSONA (codice_fiscale, nome, cognome, eta, sesso, universita)
        universita → UNIVERSITA.nome


IMPIANTO SPORTIVO (via, citta, n_campi,, contatto_segreteria, gestione)
        getione → UNIVERSITA.nome


ORARIO_IMPIANTO(impianto, giorno_settimana, apertura, chiusura)
        impianto → IMPIANTO_SPORTIVO.(via, città)


CAMPO (codice, prenotabile, responsabile, via, citta)
        responsabile → PERSONA.codice_fiscale
        via, citta → IMPIANTO_SPORTIVO.(via, citta)


SPORT (nome_sport, numero_attività)


ATTIVITA (codice_attivita, costo, sport)
        sport → SPORT.nome_sport


ACCESSI_LIMITATI (codice_abbonamento, n_accessi)
        codice_abbonamento →ATTIVITA.codice


PERIODICO (codice_abbonamento, periodo)
        codice_abbonamento →ATTIVITA.codice_attivita


CORSO (codice_corso, categoria, sesso)
        codice_corso →ATTIVITA.codice_attivita


EDIZIONE_CORSO (codice_edizione, data_inizio, data_fine, n_partecipanti, costo, corso)
        corso → CORSO.codice


LEZIONE (orario, giorno, campo, edizione_corso)
        campo → CAMPO.codice
        edizione_corso → EDIZIONE_CORSO.codice


DOCENZA (istruttore, corso)
        istruttore → PERSONA.codice_fiscale
        corso → CORSO.codice


ISCRIZIONE (studente, codice_attivita, data)
        studente → PERSONA.codice_fiscale
        codice_attivita →ATTIVITA.codice_attivita


AFFITTO(persona, campo, ora_inizio, ora fine, tariffa )
        persona → PERSONA.codice_fiscale
        campo → CAMPO.codice


ACCESSO (campo, abbonamento)
        campo → CAMPO.codice
        abbonamento → ATTIVITA.codice_attivita


COMPATIBILITA (campo, sport)
        campo → CAMPO.codice
        sport → SPORT.nome_sport
