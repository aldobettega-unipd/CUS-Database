CREATE TABLE PERSONA (
        codice_fiscale VARCHAR(16) PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    cognome VARCHAR(100) NOT NULL,
    eta INT NOT NULL,
    sesso VARCHAR(1) NOT NULL CHECK (sesso IN ('M', 'F')),
    immatricolato BOOL NOT NULL
);


CREATE TABLE TESSERA (
   persona VARCHAR(16) PRIMARY KEY,
   data DATE NOT NULL,
   costo FLOAT NOT NULL,
   FOREIGN KEY (persona) REFERENCES PERSONA(codice_fiscale)
);


CREATE TABLE IMPIANTO_SPORTIVO (
    via VARCHAR(100),
    comune VARCHAR(100),
    nome VARCHAR(100) NOT NULL,
    contatto_segreteria VARCHAR(14) NOT NULL,
    PRIMARY KEY (via, comune)
);


CREATE TABLE ORARIO_IMPIANTO (
    via VARCHAR(100),
    comune VARCHAR(100),
    giorno_settimana VARCHAR(10) CHECK (
        giorno_settimana IN ('lunedi', 'martedi', 'mercoledi', 'giovedi', 'venerdi', 'sabato', 'domenica')
    ),
    apertura TIME NOT NULL,
    chiusura TIME NOT NULL,
    PRIMARY KEY (via,comune, giorno_settimana),
    FOREIGN KEY (via,comune) REFERENCES IMPIANTO_SPORTIVO(via,comune)
);




CREATE TABLE CAMPO (
    codice VARCHAR(8) PRIMARY KEY,
    via VARCHAR(100) NOT NULL,
    comune VARCHAR(100) NOT NULL,
    tariffa_h INT,
    responsabile VARCHAR(16) NOT NULL,
    FOREIGN KEY (via, comune) REFERENCES IMPIANTO_SPORTIVO(via, comune),
    FOREIGN KEY (responsabile) REFERENCES PERSONA(codice_fiscale)
);




CREATE TABLE ATTIVITA (
    codice_attivita VARCHAR(5) PRIMARY KEY,
    costo DECIMAL(10,2) NOT NULL CHECK (costo >= 0),
    sport VARCHAR(100) NOT NULL
);


CREATE TABLE ABBONAMENTO (
    codice_abbonamento VARCHAR(5) PRIMARY KEY,
    tipo VARCHAR(20) NOT NULL CHECK (tipo IN ('limitato', 'periodico') ),
    FOREIGN KEY (codice_abbonamento) REFERENCES ATTIVITA(codice_attivita)
);




CREATE TABLE ACCESSI_LIMITATI (
    codice_abbonamento VARCHAR(5) PRIMARY KEY,
    n_accessi INT NOT NULL CHECK (n_accessi > 0),
    FOREIGN KEY (codice_abbonamento) REFERENCES ABBONAMENTO(codice_abbonamento)
);


CREATE TABLE PERIODICO (
    codice_abbonamento VARCHAR(5) PRIMARY KEY,
    periodo VARCHAR(50) NOT NULL,
    FOREIGN KEY (codice_abbonamento) REFERENCES ABBONAMENTO (codice_abbonamento)
);


CREATE TABLE CORSO (
    codice_corso VARCHAR(5) PRIMARY KEY,
    categoria VARCHAR(100),
    sesso VARCHAR(1) NOT NULL CHECK (sesso IN ('M', 'F', 'U')),
    FOREIGN KEY (codice_corso) REFERENCES ATTIVITA(codice_attivita)
);


CREATE TABLE EDIZIONE_CORSO (
    codice_edizione VARCHAR(10) PRIMARY KEY,
    data_inizio DATE NOT NULL,
    data_fine DATE NOT NULL,
    n_partecipanti INT NOT NULL CHECK (n_partecipanti > 0),
    corso VARCHAR(5) NOT NULL,
    FOREIGN KEY (corso) REFERENCES CORSO(codice_corso),
    CHECK (data_inizio < data_fine)
);


CREATE TABLE LEZIONE (
    giorno_settimana VARCHAR(10) NOT NULL CHECK (
        giorno_settimana IN ('Lunedì', 'Martedì', 'Mercoledì', 'Giovedì', 'Venerdì', 'Sabato', 'Domenica')
    ),
    orario_inizio TIME,
    orario_fine TIME NOT NULL,
    campo VARCHAR(8) NOT NULL,
    edizione_corso VARCHAR(10) NOT NULL,
    PRIMARY KEY (giorno_settimana, orario_inizio, campo),
    FOREIGN KEY (campo) REFERENCES CAMPO(codice),
    FOREIGN KEY (edizione_corso) REFERENCES EDIZIONE_CORSO(codice_edizione)
);


CREATE TABLE DOCENZA (
    istruttore VARCHAR(16) NOT NULL,
    corso VARCHAR(5) NOT NULL,
    PRIMARY KEY (istruttore, corso),
    FOREIGN KEY (istruttore) REFERENCES PERSONA(codice_fiscale),
    FOREIGN KEY (corso) REFERENCES CORSO(codice_corso)
);


CREATE TABLE ISCRIZIONE (
    studente VARCHAR(16),
    codice_attivita VARCHAR(5),
    data DATE,
    PRIMARY KEY (studente, codice_attivita, data),
    FOREIGN KEY (studente) REFERENCES PERSONA(codice_fiscale),
    FOREIGN KEY (codice_attivita) REFERENCES ATTIVITA(codice_attivita)
);


CREATE TABLE AFFITTO (
    persona VARCHAR(16),
    campo VARCHAR(8),
    data DATE,
    ora_inizio TIME,
    ora_fine TIME NOT NULL,
    PRIMARY KEY (persona, campo, data, ora_inizio),
    FOREIGN KEY (persona) REFERENCES PERSONA(codice_fiscale),
    FOREIGN KEY (campo) REFERENCES CAMPO(codice),
    CHECK (ora_inizio < ora_fine)
);


CREATE TABLE ACCESSO (
    campo VARCHAR(8),
    abbonamento VARCHAR(5),
    PRIMARY KEY (campo, abbonamento),
    FOREIGN KEY (campo) REFERENCES CAMPO(codice),
    FOREIGN KEY (abbonamento) REFERENCES ABBONAMENTO(codice_abbonamento)
);


CREATE TABLE COMPATIBILITA (
    campo VARCHAR(8),
    sport VARCHAR(100),
    PRIMARY KEY (campo, sport),
    FOREIGN KEY (campo) REFERENCES CAMPO(codice)
);
